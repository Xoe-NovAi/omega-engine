class HardwareMonitor:
    """
    Collects live hardware statistics from the system.
    
    Uses psutil for cross-platform system monitoring and gathers:
    - CPU utilization and core details
    - Memory usage (total, available, used)
    - Temperature sensors (if accessible)
    - Disk usage statistics
    - Network interface information
    """

    def __init__(self, config: HardwareConfig):
        self.config = config
        self._stats: Optional[HardwareStats] = None
        self._last_collect_time: Optional[datetime] = None

    def collect(self) -> HardwareStats:
        """
        Collect all available hardware statistics.
        
        Returns:
            HardwareStats: Structured dictionary with all collected metrics.
            
        Raises:
            HardwareError: If any step of collection fails.
        """
        try:
            # Gather each component
            cpu = self._gather_cpu()
            memory = self._gather_memory()
            temperature = self._gather_temperature()
            disk = self._gather_disk()
            network = self._gather_network()

            self._stats = HardwareStats(
                timestamp=datetime.now(),
                cpu=cpu,
                memory=memory,
                temperature=temperature,
                disk=disk,
                network=network,
            )
            self._last_collect_time = datetime.now()
            logger.info("Hardware stats collected successfully")

        except HardwareError as e:
            logger.error(f"Failed to collect hardware stats: {e}")
            raise
        except Exception as e:
            logger.error(f"Unexpected error during hardware collection: {e}")
            raise HardwareCollectionError(str(e)) from e

    def _gather_cpu(self) -> Dict[str, Any]:
        """Gather CPU information."""
        try:
            cpu_count = os.cpu_count()
            if cpu_count == 0:
                raise HardwareError("Could not determine CPU count")

            # Get CPU details via psutil
            cpu_details = psutil.CPUPctUtil()
            cpu_info = {
                "total_cores": cpu_count,
                "logical_cores": cpu_count if cpu_count > 0 else 0,
                "current_utilization_percent": cpu_details.percent if cpu_details else {},
                "max_clock_hz": cpu_details.max_freq if cpu_details else 0,
                "min_clock_hz": cpu_details.min_freq if cpu_details else 0,
                "cpu_frequencies": cpu_details.freq if cpu_details else {},
            }
            return cpu_info
        except Exception as e:
            logger.error(f"CPU gathering failed: {e}")
            raise HardwareError(f"CPU gathering failed: {e}") from e

    def _gather_memory(self) -> Dict[str, Any]:
        """Gather memory information."""
        try:
            mem = psutil.virtual_memory()
            total = mem.total
            available = mem.available
            used = mem.used
            percent = mem.percent

            # Detailed breakdown if available
            if mem.summary():
                # Get memory chunks for detailed breakdown
                pass

            return {
                "total_gb": round(total / (1024 ** 3), 2),
                "available_gb": round(available / (1024 ** 3), 2),
                "used_gb": round(used / (1024 ** 3), 2),
                "percent_used": round(percent, 2),
                "memory_type": "system" if mem.total > 0 else "unknown",
            }
        except Exception as e:
            logger.error(f"Memory gathering failed: {e}")
            raise HardwareError(f"Memory gathering failed: {e}") from e

    def _gather_temperature(self) -> Dict[str, Any]:
        """Gather temperature sensor readings."""
        try:
            temps: Dict[str, Any] = {}
            # Common thermal zones on Ryzen 7 5600u
            for sensor_path in self.config.temperature_sensors:
                try:
                    with open(sensor_path, "r") as f:
                        temp = float(f.read().strip())
                        temps[sensor_path] = temp
                except (FileNotFoundError, PermissionError, ValueError) as e:
                    logger.debug(f"Failed to read temperature sensor {sensor_path}: {e}")
                    continue

            return {"sensors": temps}
        except Exception as e:
            logger.error(f"Temperature gathering failed: {e}")
            raise HardwareError(f"Temperature gathering failed: {e}") from e

    def _gather_disk(self) -> Dict[str, Any]:
        """Gather disk usage statistics."""
        try:
            disk = psutil.disk_partitions()
            total = 0
            used = 0
            for partition in disk:
                try:
                    usage = psutil.disk_usage(partition.mountpoint)
                    total += usage.total
                    used += usage.used
                except Exception as e:
                    logger.debug(f"Failed to get disk usage for {partition}: {e}")
                    continue

            return {
                "total_gb": round(total / (1024 ** 3), 2),
                "used_gb": round(used / (1024 ** 3), 2),
                "free_gb": round((total - used) / (1024 ** 3), 2),
                "mount_points": list(partition.mountpoint for partition in disk if partition.mountpoint),
            }
        except Exception as e:
            logger.error(f"Disk gathering failed: {e}")
            raise HardwareError(f"Disk gathering failed: {e}") from e

    def _gather_network(self) -> Dict[str, Any]:
        """Gather network interface statistics."""
        try:
            net = psutil.net_if_addrs()
            interfaces: List[Dict[str, Any]] = []
            for iface in net:
                try:
                    addrs = iface.addrs
                    if addrs:
                        interfaces.append({
                            "name": iface.labels[0] if iface.labels else "unknown",
                            "binds": [a.addr for a in addrs],
                            "group": iface.group,
                            "mac": iface.mac,
                        })
                except Exception as e:
                    logger.debug(f"Failed to parse network interface {iface}: {e}")
                    continue

            return {
                "interfaces": interfaces,
                "total_connections": sum(
                    len(iface.addrs) for iface in net
                ),
                "connected_interfaces": len(interfaces),
            }
        except Exception as e:
            logger.error(f"Network gathering failed: {e}")
            raise HardwareError(f"Network gathering failed: {e}") from e

class HardwareProvider(MCPProvider):
    """
    MCP Provider that exposes hardware monitoring capabilities.
    
    Implements the MCP provider interface to allow external systems
    to query hardware statistics through the MCP protocol.
    """

    def __init__(self, monitor: HardwareMonitor):
        super().__init__()
        self.monitor = monitor

    async def query(self, query: str) -> MCPQueryResult:
        """
        Execute a query against the MCP server.
        
        Args:
            query: The query string to send to the MCP server.
            
        Returns:
            MCPQueryResult: Contains query_id, status, data, and optional error.
        """
        query_id = f"hardware_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
        
        try:
            # Send query to MCP server
            response = await self.client.query(query)
            return MCPQueryResult(
                query_id=query_id,
                status="success",
                data=response,
                error=None,
            )
        except MCPProviderError as e:
            logger.error(f"MCP query failed: {e}")
            return MCPQueryResult(
                query_id=query_id,
                status="error",
                data=None,
                error=str(e),
            )
        except Exception as e:
            logger.error(f"Unexpected error during MCP query: {e}")
            return MCPQueryResult(
                query_id=query_id,
                status="error",
                data=None,
                error=str(e),
            )

class HardwareAgent:
    """
    Main agent that orchestrates hardware monitoring and MCP integration.
    
    This agent:
    - Initializes the MCP client connection
    - Periodically collects hardware statistics
    - Exposes them via MCP provider queries
    - Handles graceful shutdown and error recovery
    """

    def __init__(
        self,
        mcp_config: MCPConfig,
        hardware_config: HardwareConfig,
    ):
        self.mcp_config = mcp_config
        self.hardware_config = hardware_config
        self.monitor = HardwareMonitor(hardware_config)
        self.provider = HardwareProvider(self.monitor)
        self._client: Optional[MCPClient] = None
        self._running = False

    async def initialize(self) -> None:
        """Initialize the agent and establish MCP connection."""
        logger.info("Initializing Hardware Agent...")
        
        # Connect to MCP server
        try:
            self._client = MCPClient(self.mcp_config.server_url)
            await self._client.connect()
            logger.info(f"Connected to MCP server at {self.mcp_config.server_url}")
        except Exception as e:
            logger.error(f"Failed to connect to MCP server: {e}")
            raise

    async def collect_and_expose(self) -> HardwareStats:
        """
        Collect hardware stats and expose them via MCP.
        
        Returns:
            HardwareStats: The collected statistics.
        """
        logger.info("Collecting hardware statistics...")
        stats = await self.monitor.collect()
        logger.info("Hardware stats collected")
        return stats

    async def run(self) -> None:
        """Main loop: collect stats periodically and expose via MCP."""
        while self._running:
            try:
                # Collect hardware stats
                stats = await self.collect_and_expose()
                
                # Expose via MCP (example: return stats as a query result)
                await self.provider.query("get_hardware_stats")
                
                # Also expose as a direct MCP query
                await self.provider.query("hardware.stats")
                
                # Log the latest stats
                logger.info(
                    f"Latest hardware stats: "
                    f"CPU={stats.cpu.get('total_cores', 'N/A')}, "
                    f"Mem={stats.memory.get('used_gb', 'N/A')}GB, "
                    f"Temp={stats.temperature.get('sensors', {}).get('thermal_zone0', 'N/A')}"
                )
                
            except MCPProviderError as e:
                logger.error(f"MCP query failed: {e}")
                # In a production system, you might want to implement retry logic here
            except HardwareError as e:
                logger.error(f"Hardware collection failed: {e}")
            except Exception as e:
                logger.error(f"Unexpected error in main loop: {e}")
            
            # Wait before next collection (respect interval)
            await anyio.sleep(self.hardware_config.stats_interval_seconds)

    def stop(self) -> None:
        """Gracefully stop the agent."""
        self._running = False
        if self._client:
            await self._client.disconnect()
        logger.info("Hardware Agent stopped")
