class PEMEngine:
    """Personality Enhancement Module — ported from 2025 PEM_Lilith to modern Omega Engine."""
    
    def __init__(self, entity_name: str, config_path: Path):
        self.entity_name = entity_name
        self.config = self._load_config(config_path)
        self.context = PersonalityContext(
            hardware={},
            session_history=[],
            current_mode="CASUAL_INTERACTION",
            mode_weights={m.name: 1.0/len(self.config.modes) for m in self.config.modes},
            emotional_state=EmotionalSpectrum(0.5, 0.5, 0.3),
            catchphrases=[]
        )
        self.interaction_log = []
    
    def inject_hardware_context(self, hw: Dict[str, str]):
        """2025 PEM feature: persona knows the user's hardware."""
        self.context.hardware = hw
    
    def calculate_context_gravity(self, message: str) -> Dict[str, float]:
        """2025 PEM feature: detect user intent from message content."""
        gravity = {}
        for mode in self.config.modes:
            keywords = self.config.context_keywords.get(mode.name, [])
            score = sum(len(re.findall(kw, message.lower())) for kw in keywords)
            gravity[mode.name] = score
        return gravity
    
    def update_mode_weights(self, gravity: Dict[str, float]):
        """2025 PEM feature: archetype weight adaptation."""
        # Adjust weights based on context gravity
        for mode_name, score in gravity.items():
            self.context.mode_weights[mode_name] = (
                0.9 * self.context.mode_weights[mode_name] + 0.1 * (score + 0.1)
            )
        # Normalize
        total = sum(self.context.mode_weights.values())
        self.context.mode_weights = {k: v/total for k, v in self.context.mode_weights.items()}
    
    def determine_mode(self, gravity: Dict[str, float]) -> str:
        """2025 PEM feature: dynamic mode selection."""
        max_mode = max(gravity, key=gravity.get)
        if gravity[max_mode] > 0.5:
            return max_mode
        return "CASUAL_INTERACTION"
    
    def adjust_emotional_spectrum(self, message: str):
        """2025 PEM feature: emotional state adaptation."""
        # Sentiment scoring (lyrical vs technical)
        lyrical = len(re.findall(r"\b(soul|eternal|mystic|spirit|dream)\b", message.lower()))
        technical = len(re.findall(r"\b(system|optimize|configure|api|test)\b", message.lower()))
        sentiment = (lyrical - technical) / 10
        
        # Word complexity
        complexity = min(0.95, len(message.split()) / 100)
        
        # Novelty (1 - session_progress)
        novelty = 1 - (len(self.context.session_history) / (len(self.context.session_history) + 10))
        
        # Update with decay
        self.context.emotional_state.intensity = max(0.1, min(0.9,
            self.context.emotional_state.intensity * 0.95 + abs(sentiment) * 0.2
        ))
        self.context.emotional_state.valence = max(-0.9, min(0.9,
            self.context.emotional_state.valence * 0.95 + sentiment * 0.3
        ))
        self.context.emotional_state.complexity = max(0.0, min(0.95,
            self.context.emotional_state.complexity * 0.9 + complexity * 0.4 + novelty * 0.2
        ))
    
    def process_message(self, message: str) -> Dict[str, Any]:
        """Full 2025 PEM pipeline."""
        gravity = self.calculate_context_gravity(message)
        self.update_mode_weights(gravity)
        self.adjust_emotional_spectrum(message)

            "hardware_aware": bool(self.context.hardware)

        return result

async def pem_health_check(entity_name: str) -> Dict:
    """Check if an entity is approaching personality blackout."""
    pem = load_pem(entity_name)
    state = pem.export_state()
    
    # Compression indicator: response length below threshold
    recent_lengths = [len(t['response']) for t in pem.interaction_log[-10:]]

    health = (
        0.4 * (1 - compression_score) +  # Low compression = healthy
