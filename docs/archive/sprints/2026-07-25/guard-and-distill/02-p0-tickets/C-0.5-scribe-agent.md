## Implementation Sketch (Self-Contained)

```python
# File: src/omega/scribe/__init__.py
# Purpose: Scribe agent — L1→L2→L3 distillation pipeline
# Dependencies: anyio, pydantic, jsonschema, src.omega.memory_store, src.omega.oracle

import anyio
from pydantic import BaseModel, Field
from typing import List, Optional, Dict, Any
from src.omega.memory_store import MemoryStore
from src.omega.oracle import Oracle

class L1Narrative(BaseModel):
    session_id: str; entity_name: str
    exchanges: List[Dict[str, Any]]; summary: str
    token_count: int; timestamp: str

class L2Insight(BaseModel):
    narrative_id: str; insights: List[Dict[str, Any]]
    themes: List[str]; personalization_depth: float

class L3Axiom(BaseModel):
    insight_id: str; axiom: str; domain: str
    confidence: float; json_schema: Dict[str, Any]
    source_evidence: List[str]; cross_pollination_targets: List[str]

class ProposedLesson(BaseModel):
    entity_name: str; session_id: str
    l1_narrative: L1Narrative; l2_insights: List[L2Insight]
    l3_axioms: List[L3Axiom]; created_at: str
    status: str = "staged"; promoted_at: Optional[str] = None
    promoted_by: Optional[str] = None

class Scribe:
    def __init__(self, memory_store: MemoryStore, oracle: Oracle):
        self.memory = memory_store; self.oracle = oracle
        self.l1_model = "local:qwen3-4b-thinking"
        self.l2_model = "local:qwen3-4b-thinking"
        self.l3_model = "local:nemotron-3-ultra"
    
    async def distill_session(self, session_id: str, entity_name: str) -> ProposedLesson:
        exchanges = await self.memory.get_session(session_id, entity_name)
        l1 = await self._distill_l1_narrative(exchanges, entity_name, session_id)
        l2_insights = await self._distill_l2_insights(l1, entity_name)
        l3_axioms = await self._distill_l3_axioms(l2_insights, entity_name)
        lesson = ProposedLesson(entity_name=entity_name, session_id=session_id,
            l1_narrative=l1, l2_insights=l2_insights, l3_axioms=l3_axioms,
            created_at=anyio.current_time().isoformat())
        await self._stage_lesson(entity_name, lesson)
        await self._suggest_cross_pollination(entity_name, l3_axioms)
        return lesson
    
    async def _distill_l1_narrative(self, exchanges, entity_name, session_id) -> L1Narrative:
        prompt = f"""Distill this {entity_name} session into a coherent narrative.
Focus on: decisions made, patterns discovered, friction points, outcomes.
Preserve entity voice and context. Output as structured JSON."""
        result = await self.oracle.summon_local(entity="scribe",
            query=prompt + "\n\n" + self._format_exchanges(exchanges), model=self.l1_model)
        return L1Narrative(session_id=session_id, entity_name=entity_name,
            exchanges=exchanges, summary=result, token_count=len(result.split()),
            timestamp=anyio.current_time().isoformat())
    
    async def _distill_l2_insights(self, l1: L1Narrative, entity_name: str) -> List[L2Insight]:
        prompt = f"""From this narrative, extract actionable insights.
Each insight must have: claim, evidence (quote/ref), confidence (0-1), tags.
Focus on: what worked, what failed, what's reusable, what's entity-specific."""
        result = await self.oracle.summon_local(entity="scribe",
            query=prompt + "\n\nNarrative: " + l1.summary, model=self.l2_model)
        return self._parse_insights(result)
    
    async def _distill_l3_axioms(self, l2_insights: List[L2Insight], entity_name: str) -> List[L3Axiom]:
        axiom_schema = {"type": "object", "properties": {
            "axiom": {"type": "string"}, "domain": {"type": "string", "enum": ["sovereignty","memory","coordination","inference","architecture","heritage","security"]},
            "confidence": {"type": "number", "minimum": 0, "maximum": 1},
            "source_evidence": {"type": "array", "items": {"type": "string"}},
            "cross_pollination_targets": {"type": "array", "items": {"type": "string"}}
        }, "required": ["axiom","domain","confidence","source_evidence"], "additionalProperties": False}
        prompt = f"""Extract universal principles (L3 axioms) from these insights.
Each axiom must be: timeless, domain-specific, evidence-grounded, actionable.
Output as JSON array matching the provided schema.
Self-critique: verify each axiom is not entity-specific, not situational, not trivial."""
        result = await self.oracle.summon_local(entity="scribe",
            query=prompt + "\n\nInsights: " + str([i.dict() for i in l2_insights]),
            model=self.l3_model, json_schema=axiom_schema)
        return [L3Axiom(**a, insight_id=f"l3_{i}") for i, a in enumerate(result)]
    
    async def _stage_lesson(self, entity_name: str, lesson: ProposedLesson):
        import yaml; from pathlib import Path
        path = Path(f"data/entities/{entity_name}/proposed_lessons.yaml")
        path.parent.mkdir(parents=True, exist_ok=True)
        tmp_path = path.with_suffix(".yaml.tmp")
        existing = yaml.safe_load(path.read_text()) or [] if path.exists() else []
        existing.append(lesson.dict())
        tmp_path.write_text(yaml.dump(existing, sort_keys=False))
        tmp_path.rename(path)
    
    async def _suggest_cross_pollination(self, entity_name: str, axioms: List[L3Axiom]):
        for axiom in axioms:
            for target in axiom.cross_pollination_targets:
                if target != entity_name:
                    suggestion = {"type": "cross_pollination_suggestion", "from_entity": entity_name,
                        "axiom": axiom.axiom, "domain": axiom.domain, "confidence": axiom.confidence,
                        "suggested_at": anyio.current_time().isoformat()}
                    await self._append_suggestion(target, suggestion)

async def session_end_hook(session_id: str, entity_name: str):
    from src.omega.memory_store import MemoryStore
    from src.omega.oracle import Oracle
    memory = MemoryStore(); oracle = Oracle(); scribe = Scribe(memory, oracle)
    lesson = await scribe.distill_session(session_id, entity_name)
    live_feed = Path(f"data/coordination/SCRIBE_LIVE_FEED.md")
    live_feed.parent.mkdir(parents=True, exist_ok=True)
    with live_feed.open("a") as f:
        f.write(f"[{anyio.current_time().isoformat()}] SCRIBE DISTILLED {entity_name}/{session_id} → {len(lesson.l3_axioms)} L3 axioms staged\n")
```
